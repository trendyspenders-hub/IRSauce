cmake_minimum_required(VERSION 3.22)
project(IRSauce VERSION 0.3.0)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(FORMATS VST3)
if(APPLE)
 list(APPEND FORMATS AU)
endif()
juce_add_binary_data(SauceAssets SOURCES Assets/Smoker.jpg)
juce_add_plugin(IRSauce COMPANY_NAME "I.R.Audio" BUNDLE_ID audio.ir.sauce PLUGIN_MANUFACTURER_CODE IrAu PLUGIN_CODE IrSc FORMATS ${FORMATS} PRODUCT_NAME "IRSauce" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(IRSauce PRIVATE Source/Plugin.cpp Source/Editor.cpp)
target_include_directories(IRSauce PRIVATE vendor/signalsmith-stretch vendor/signalsmith-stretch/include)
target_compile_definitions(IRSauce PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(IRSauce PRIVATE SauceAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)

include(CTest)
if(BUILD_TESTING)
 add_executable(sauce_core_tests tests/core_test.cpp)
 target_include_directories(sauce_core_tests PRIVATE vendor/signalsmith-stretch vendor/signalsmith-stretch/include)
 add_test(NAME sauce_core COMMAND sauce_core_tests)
endif()
if(BUILD_TESTING)
 add_executable(sauce_processor_tests tests/processor_test.cpp)
 target_link_libraries(sauce_processor_tests PRIVATE IRSauce)
 target_include_directories(sauce_processor_tests PRIVATE $<TARGET_PROPERTY:IRSauce,INCLUDE_DIRECTORIES>)
 target_compile_definitions(sauce_processor_tests PRIVATE $<TARGET_PROPERTY:IRSauce,COMPILE_DEFINITIONS>)
 add_test(NAME sauce_processor COMMAND sauce_processor_tests)
endif()
